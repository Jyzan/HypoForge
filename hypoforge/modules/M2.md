# M2 文献检索与知识提取 — 类函数伪代码

> 对应代码文件：`m2_literature_search.py`
> 类名：`M2LiteratureSearch`
>
> **模块职责一句话**：输入 M1 的问题卡片（子问题 + 关键实体），去 PubMed / OpenAlex 检索论文，
> 用千问把论文摘要提取成结构化知识条目，输出 `LiteratureResult` 列表交给 M3。
>
> **为什么全是异步函数**：检索和调千问都是"等网络"，异步可以让多个请求同时进行（`asyncio.gather`），
> 用 `Semaphore` 限流防止触发 API 限速。

---

## 函数总览（调用关系）

```
Pipeline 调用
   │
   ▼
__call__（总调度）
   ├─ 首轮 ──────→ _run_real
   │                ├─ _generate_search_queries（中文子问题 → 英文检索词）
   │                ├─ 双后端（PubMed + OpenAlex）并行搜索
   │                ├─ 合并去重，取 Top-N 篇论文
   │                └─ 分批并发 _extract_batch（批量提取知识）
   │                    └─（batch_size=1 时退化为 _extract_from_paper）
   └─ 补搜轮 ────→ _run_supplement（缓存优先的增量补搜）
```

## 1. `__init__` — 初始化（构造函数）

```python
def __init__(self, search_tools, mode, llm_config, query_llm_config,
             max_papers_per_query, batch_size, max_search_queries,
             query_max_tokens, query_disable_thinking,
             query_max_attempts, supplement_paper_budget, ...):
    # 纯配置赋值，不干别的活：
    self.search_tools = 指定的检索工具列表（默认 ["semantic_scholar", "pubmed"]）
    self.mode = 运行模式（默认 "llm"）
    self.max_papers_per_query = 每个检索词最多保留几篇论文（默认 10）
    self.batch_size = 每批打包几篇论文一起提取（默认 5，最小 1）
    self.max_search_queries = 每个子问题生成几个英文检索词（0 = 不生成，直接用原问题）
    self.query_max_tokens / query_disable_thinking / query_max_attempts = 生成检索词时的 LLM 参数
    self.supplement_paper_budget = 补搜轮每轮的论文预算上限（默认 6）

    # 创建两个千问客户端：
    self.client = 主客户端（负责知识提取）
    self.query_client = 生成检索词专用客户端（单独一个，因为便宜档模型翻译容易返回空，
                        换更可靠的模型档来干这个活）
```

## 2. `__call__` — 总调度员（模块唯一入口）

```python
async def __call__(self, state, config=None):
    """Pipeline 只调用这一个函数，它负责分流。"""

    # ---- 第一步：判断是不是"补搜轮"（M6 回跳过来的）----
    open_gaps = 从 state 里挑出所有状态为 "open"（未解决）的证据缺口

    if state.search_round > 0 且 state.literature_results 非空:
        # 这是补搜轮：
        if 没有 open_gaps:
            # 没有缺口要补 → 别白跑，原样把已有结果返回
            return {literature_results: 原封不动的已有结果}

        if 有 open_gaps:
            # 有缺口 → 转交给 _run_supplement 做增量补搜
            return await self._run_supplement(state, open_gaps)

    # ---- 第二步：首轮正式检索 ----
    if self.client 是 None:
        报错: "M2 需要 LLM 客户端，请配置 OPENAI_API_KEY"

    sub_questions = 从 state.problem_card 取子问题（没有卡片就用原始问题兜底）
    key_entities = 从 state.problem_card 取关键实体（没有就给空列表）

    results, collected = await self._run_real(sub_questions, key_entities)

    # 顺手把论文写进持久化缓存（PaperStore），供以后补搜轮命中用（只做副作用，不改结果）
    if state 配置了 memory_cache_dir:
        for 每个子问题的 (query, papers):
            把 papers 存进 PaperStore

    return {literature_results: results}
```

**一句话**：先判断是首轮还是补搜轮 → 首轮走 `_run_real`，补搜轮走 `_run_supplement`。

## 3. `_run_real` — 正式检索主流程

```python
async def _run_real(self, sub_questions, key_entities):
    """对每一个子问题，完整跑一遍：翻译 → 搜索 → 合并 → 提取。"""

    pubmed = PubMedTool()          # 生物医学文献库
    academic = SemanticScholarTool()  # 学术文献库（OpenAlex 后端）

    results = []    # 最终返回给 state 的结果
    collected = []  # 内部用的记录（存缓存用，不进 state）

    for sq in sub_questions:   # 循环：对每个子问题
        # ① 生成英文检索词（中文子问题 PubMed 搜不到东西）
        queries = await self._generate_search_queries(sq, key_entities)

        # ② 每个检索词 × 两个后端，同时发起搜索
        search_calls = []
        for q in queries:
            search_calls.append(("pubmed", pubmed.search(q, limit=max_papers)))
            search_calls.append(("academic", academic.search(q, limit=max_papers)))

        raw_results = await asyncio.gather(*全部搜索请求, return_exceptions=True)
        # 哪个请求失败了就记个 warning 跳过，不拖垮整个流程

        # ③ 合并 PubMed 和学术库的结果，去重，取前 N 篇
        pubmed_papers = 收集 pubmed 的结果
        acad_papers = 收集 academic 的结果
        all_papers = _merge_deduplicate(pubmed_papers, acad_papers)
        top_papers = all_papers[:max_papers_per_query]

        # ④ 按 batch_size 分组，并发批量提取知识
        all_entries = []
        if top_papers 非空:
            batches = 把 top_papers 切成 [每批5篇] 的列表
            sem = asyncio.Semaphore(3)   # 同时最多 3 个 LLM 调用，防止限流

            entry_lists = await asyncio.gather(
                *[对每个 batch 执行（在 sem 保护下调用 _extract_batch）],
                return_exceptions=True,
            )
            for entries in entry_lists:
                if 提取失败: 记 warning 跳过
                else: all_entries 合并这批结果

        # ⑤ 打包成本子问题的结果
        results.append(LiteratureResult(
            sub_question=sq,
            papers_retrieved=len(top_papers),
            knowledge_entries=all_entries,
        ))
        collected.append((sq, queries, top_papers))

    return results, collected
```

**一句话**：`for 子问题 → 翻译 → 并发搜 → 去重 → 并发提取 → 组装结果`。

## 4. `_generate_search_queries` — 翻译官（中文问题 → 英文检索词）

```python
async def _generate_search_queries(self, sub_question, key_entities):
    """让千问把（可能是中文的）子问题改写成几个聚焦的英文检索词。"""

    # 没配置生成客户端，或配置要求不生成 → 直接用原问题去搜
    if self.query_client 是 None 或 max_search_queries <= 0:
        return [sub_question]

    user_prompt = 按模板拼出提示词（带上子问题 + 关键实体列表）

    for attempt in range(1, query_max_attempts + 1):   # 重试循环
        try:
            raw_text = await self.query_client.chat(
                system_prompt=模板,
                user_prompt=user_prompt,
                disable_thinking=True,   # 纯翻译任务，不让模型想太多
            )
        except 调用失败:
            记 warning，继续重试

        cleaned = _extract_queries_from_text(raw_text)   # 从自由文本里解析出检索词列表
        if cleaned 非空:
            return cleaned[:max_search_queries]   # 成功，截断到上限返回

        # 返回了空内容 → 再试一次

    # 重试全失败：报错！拒绝"静默用中文原问题去搜"
    raise RuntimeError("多次尝试后仍无法生成英文检索词，拒绝搜索")
```

**一句话**：调千问把子问题翻译成英文检索词，失败重试，全失败直接报错不糊弄。

## 5. `_extract_batch` — 批量知识提取（主力）

```python
async def _extract_batch(self, papers):
    """把一批（默认5篇）论文打包成一次 LLM 调用，提取所有知识条目。"""

    if papers 为空: return []

    # ① 给每篇论文造一个稳定 ID（有 PMID 用 "PMID:xxx"，否则用 paper_id，再没有用序号）
    paper_map = {}
    paper_blocks = []
    for idx, paper in enumerate(papers):
        source_id = f"PMID:{pmid}" if pmid 存在 else (paper_id or f"IDX{idx}")
        paper_map[source_id] = paper
        paper_blocks.append(把这篇论文的标题/作者/年份/期刊/摘要按模板格式化)

    # ② 一次 LLM 调用，让千问输出所有知识条目（JSON 数组）
    user_prompt = 模板.format(paper_count=len(papers), papers_text=所有论文文本)
    raw_entries = await self.client.structured_chat(
        system_prompt=提取专家提示词,
        user_prompt=user_prompt,
        output_schema=JSON格式（type/content/confidence/source_paper_id/entities）,
        max_tokens=16384,   # 提取结果可能很大
    )

    # ③ 解析 + 规范化成 KnowledgeEntry
    if 返回的是 dict: 从中取 entries/data/items 里的列表

    entries = []
    for raw in raw_entries:
        source_id = raw 里的来源论文 ID
        source_title = raw 里的论文标题（LLM 没给就从 paper_map 里查）

        entries.append(KnowledgeEntry(
            id=_stable_entry_id(内容, source_id),   # 用"内容+论文ID"打稳定ID（指纹）
            type=知识类型（established_fact 等六类之一）,
            content=知识内容,
            confidence=置信度（high/medium/low，可空）,
            source_paper_id=来源论文 ID,
            source_paper_title=来源论文标题,
            entities=涉及实体列表,
        ))

    return entries
```

**一句话**：多篇论文一次调用提取，省 LLM 调用次数（15 篇 → 3 次而不是 15 次）。

## 7. `_run_supplement` — 补搜执行者（M6 回跳时用）

```python
async def _run_supplement(self, state, open_gaps):
    """缓存优先的增量补搜：能命中缓存就不重新检索，没命中才实时搜索。"""

    store = 打开持久化论文缓存（PaperStore）
    merged_results = 深拷贝现有 literature_results（绝不清空旧结果！）
    existing_sub_questions = 已有结果的子问题列表
    known_keys = 已见过的论文 key 集合（来自 search_ledger）
    issued_norms = 已发过的检索词集合（规范化后）
    remaining_budget = supplement_paper_budget   # 本轮补搜预算（默认6篇）
    new_query_texts = []   # 新发出的检索词
    new_paper_keys = []    # 新见过的论文
    attempted_gap_ids = set()   # 处理过的缺口 ID

    for gap in open_gaps:   # 循环：逐个缺口补

        sub_question = 解析这个缺口属于哪个子问题
        candidates = 这个缺口建议的候选检索词

        # ---- 第一步：先查缓存 ----
        if store 存在:
            hit_keys = 用候选检索词查缓存，找到之前存过、且本轮还没见过的论文 key
            if 有命中的:
                # 命中 → 不提取知识（那批论文早提取过了），只把论文计数加上去
                merge_literature_increment(merged_results, sub_question, len(命中的), 无新条目)
                known_keys 更新
                记一个 "memory_hit" 事件（前端能看到：命中缓存 X 篇）
                continue   # 这个缺口处理完了，下一个

        # ---- 第二步：缓存没命中 → 实时搜索 ----
        attempted_gap_ids.add(gap.gap_id)
        fresh_queries = 过滤掉已发过的检索词（dedupe）
        if 没有新检索词 或 预算用完:
            continue   # 没得搜了，跳过

        pubmed / academic = 懒加载创建两个检索工具
        search_calls = 每个新检索词 × 两个后端
        raw_results = await asyncio.gather(*所有搜索)   # 并发

        all_papers = _merge_deduplicate(pubmed 结果, academic 结果)
        new_papers = 去掉已见过的（按 key），再截断到剩余预算
        if store 存在:
            把新检索词和新论文写回缓存
        if 没有新论文:
            continue

        # 对新论文批量提取知识（和 _run_real 一样的并发提取逻辑）
        entries = 分批并发调用 _extract_batch 的结果汇总

        # 把新论文数 + 新知识条目合并进结果
        merge_literature_increment(merged_results, sub_question, len(new_papers), entries)
        new_paper_keys 更新
        remaining_budget -= len(new_papers)   # 扣预算

    # ---- 第三步：更新缺口状态和检索台账 ----
    updated_gaps = []
    for gap in state.evidence_gaps:
        if gap 被处理过 且 状态还是 "open":
            状态改为 "pending_grounding"   # 补完了，等 M3 去接地
        else:
            原样保留

    ledger = SearchLedger(旧检索词 + 新检索词, 旧论文 key + 新论文 key)

    return {
        literature_results: merged_results,   # 全量返回，旧结果 + 增量
        evidence_gaps: updated_gaps,          # 状态更新后的缺口
        search_ledger: ledger,                # 更新的台账
    }
```

**一句话**：先翻缓存（命中只加计数、不造假知识），翻不到才实时搜，全程受预算限制，旧结果永远不清空。

## 附：被本类调用的模块级辅助函数（不在类里，但在同一个文件）

```python
def _stable_entry_id(content, source_paper_id):
    # 知识条目的"指纹 ID"：内容 + 来源论文ID 拼起来做 SHA1 哈希，取前12位
    # 特点：同一内容永远算出同一个 ID → M3 增量去重认它
    return "KE_" + sha1(规范化内容 + 论文ID) 前12位

def _extract_queries_from_text(raw_text):
    # 从千问的自由文本回复里解析出检索词列表
    # 先试 JSON 解析（老版本兼容）→ 再试乱码抢救 → 最后按"一行一个词"逐行解析
    # 去掉序号/项目符号/引号等噪音，过滤太短的，去重后返回

def _doi_key(paper):
    # 把论文的 DOI "洗"成统一格式：转小写、去空格、剥掉
    # "https://doi.org/" / "doi:" / "doi " 前缀
    # 用途：不同数据源返回的 DOI 格式不同，洗完后才能正确比对去重

def _merge_deduplicate(pubmed_papers, academic_papers):
    # 合并两个数据源的结果：
    #   for 每篇论文 in (PubMed + 学术):
    #       如果 DOI 见过 → 跳过（DOI 去重）
    #       如果标题前80字符见过 → 跳过（标题兜底去重）
    #       否则加入结果
    # 最后按引用数降序、年份降序排序
    return 合并去重后的论文列表
```
